# ===== 状态库（幂等去重） =====

def _init_state_db():
    """初始化状态数据库"""
    STATE_DB.parent.mkdir(parents=True, exist_ok=True)
    conn = sqlite3.connect(str(STATE_DB))
    conn.execute("""
        CREATE TABLE IF NOT EXISTS tugou_daily_cache (
            trade_date TEXT PRIMARY KEY,
            card_json TEXT,
            pushed INTEGER DEFAULT 0,
            created_at TEXT,
            updated_at TEXT
        )
    """)
    conn.commit()
    conn.close()


def _is_cached(trade_date: str) -> bool:
    """检查是否已缓存"""
    if not STATE_DB.exists():
        return False
    conn = sqlite3.connect(str(STATE_DB))
    row = conn.execute(
        "SELECT 1 FROM tugou_daily_cache WHERE trade_date=? AND pushed=1",
        (trade_date,)
    ).fetchone()
    conn.close()
    return row is not None


def _save_cache(trade_date: str, card_json: str):
    """保存缓存"""
    STATE_DB.parent.mkdir(parents=True, exist_ok=True)
    conn = sqlite3.connect(str(STATE_DB))
    now = datetime.now().isoformat()
    conn.execute("""
        INSERT OR REPLACE INTO tugou_daily_cache (trade_date, card_json, pushed, created_at, updated_at)
        VALUES (?, ?, 1, ?, ?)
    """, (trade_date, card_json, now, now))
    conn.commit()
    conn.close()


# ===== 辅助函数 =====

def _safe_float(val, default=None):
    """安全转float"""
    if val is None or (isinstance(val, float) and np.isnan(val)):
        return default
    try:
        return float(val)
    except (ValueError, TypeError):
        return default


def _fmt_pct(val, digits=2, sign=True):
    """格式化百分比"""
    if val is None:
        return "N/A"
    prefix = "+" if sign and val >= 0 else ""
    return f"{prefix}{val:.{digits}f}%"


def _fmt_bp(val, digits=1):
    """格式化bp"""
    if val is None:
        return "N/A"
    prefix = "+" if val >= 0 else ""
    return f"{prefix}{val:.{digits}f}bp"


def _get_ma20_position(close: float, ma20: float) -> str:
    """判断K线与20日均线位置关系"""
    if close is None or ma20 is None:
        return "⚠️"
    if close > ma20 * 1.02:
        return "<font color='green'>上方</font>"
    elif close > ma20:
        return "<font color='green'>刚穿</font>"
    elif close > ma20 * 0.98:
        return "<font color='orange'>贴近</font>"
    else:
        return "<font color='red'>下方</font>"


def _iv_zone(iv: float) -> Tuple[str, str]:
    """IV区间判断 → (区间, 信号)"""
    if iv is None:
        return "N/A", "N/A"
    if iv < 10:
        return "<10 极低", "🟢 极度平稳"
    elif iv <= 20:
        return "10-20", "🟢 稳定区间"
    elif iv <= 23:
        return "20-23", "🟡 轻度紧张"
    elif iv <= 25:
        return "23-25", "🟡 恐慌情绪"
    elif iv <= 30:
        return "25-30", "🔴 恐慌加剧"
    else:
        return ">30", "🔴 极端恐慌"

def fetch_index_data(trade_date: str, lookback: int = 60) -> Dict[str, Dict]:
    """获取主要指数近N日数据"""
    # 计算日期范围
    td = datetime.strptime(trade_date, "%Y%m%d")
    start_date = (td - timedelta(days=lookback * 2)).strftime("%Y%m%d")  # 多取一些交易日
    
    results = {}
    for name, ts_code in TS_INDEX_CODES.items():
        try:
            df = get_stock_daily(ts_code=ts_code, start_date=start_date, end_date=trade_date)
            if df is None or df.empty:
                print(f"  ⚠️ {name} 数据为空")
                results[name] = {}
                continue
            
            # 确保日期排序
            df = df.sort_values('trade_date')
            
            # 计算指标
            close = _safe_float(df.iloc[-1].get('close'))
            pre_close = _safe_float(df.iloc[-1].get('pre_close'))
            vol = _safe_float(df.iloc[-1].get('vol'))  # 万手
            amount = _safe_float(df.iloc[-1].get('amount'))  # 千元
            
            # 涨跌幅
            change_pct = _safe_float(df.iloc[-1].get('pct_chg'))
            if change_pct is None and close and pre_close:
                change_pct = (close / pre_close - 1) * 100
            
            # 20日均线
            if len(df) >= 20:
                ma20 = df['close'].iloc[-20:].mean()
            else:
                ma20 = close
            
            # 前一日成交量
            prev_vol = _safe_float(df.iloc[-2].get('vol')) if len(df) >= 2 else None
            vol_change = ""
            if vol and prev_vol and prev_vol > 0:
                vol_pct = (vol / prev_vol - 1) * 100
                if vol_pct > 5:
                    vol_change = f"放量{vol_pct:.0f}%"
                elif vol_pct < -5:
                    vol_change = f"缩量{abs(vol_pct):.0f}%"
                else:
                    vol_change = "量平"
            
            results[name] = {
                'close': close,
                'change_pct': change_pct,
                'ma20': _safe_float(ma20),
                'ma20_position': _get_ma20_position(close, _safe_float(ma20)),
                'vol': vol,
                'amount': amount,
                'vol_change': vol_change,
            }
            print(f"  ✅ {name}: 收盘{close} 涨跌{_fmt_pct(change_pct)} 均线位置{results[name]['ma20_position']}")
        except Exception as e:
            print(f"  ❌ {name} 获取失败: {e}")
            results[name] = {}
    
    return results


def fetch_cd_rate(trade_date: str) -> Dict:

        latest_date = data.get('latest_date', '')
        
        # 历史变化
        history = data.get('history')
        prev_rate = None
        if history is not None and not history.empty:
            if len(history) >= 2:
                prev_rate = _safe_float(history.iloc[-2].get('1Y'))
        
        change_bp = None
        if rate_1y and prev_rate:
            change_bp = (rate_1y - prev_rate) * 100  # 转为bp
        
        return {
            'rate': rate_1y,
            'date': latest_date,
            'change_bp': change_bp,
        }
    except Exception as e:
        print(f"  ❌ 存单利率获取失败: {e}")
        return {}

def fetch_omo(trade_date: str) -> Dict:
    """获取央行OMO操作 — data_hub → 美元环流(meiyuanhuanliu.com) 兜底"""
    # Level 1: data_hub
    try:
        data = get_central_bank_omo(trade_date=trade_date)
        if data and data.get('net_injection') is not None:
            return {
                'net_injection': _safe_float(data.get('net_injection')),
                'outstanding_balance': _safe_float(data.get('outstanding_balance')),
                'operations': data.get('operations', {}),
            }
    except Exception:
        pass
    
    # Level 2: 美元环流 meiyuanhuanliu.com (最可靠的OMO数据源)
    try:
        import requests as _req
        import re as _re
        from datetime import datetime as _dt, timedelta as _td
        from collections import defaultdict as _dd
        
        url = "https://www.meiyuanhuanliu.com/pbc/"
        resp = _req.get(url, timeout=15, headers={'User-Agent': 'Mozilla/5.0'})
        if resp.status_code == 200:
            html = resp.text
            
            # 解析repoData中的逆回购操作数据
            repo_match = _re.search(r'var repoData\s*=\s*\{(.*?)\};\s*\n', html, _re.DOTALL)
            if repo_match:
                raw = repo_match.group(1)
                
                def _parse_str_arr(s):
                    return _re.findall(r'"([^"]*)"', s)
                def _parse_num_arr(s):
                    if not s: return []
                    res = []
                    for v in s.split(','):
                        v = v.strip()
                        if v in ('null', ''): res.append(None)
                        else:
                            try: res.append(float(v))
                            except: res.append(None)
                    return res
                
                dates = _parse_str_arr(_re.search(r'dates:\s*\[([^\]]*)\]', raw).group(1))
                term7 = _parse_num_arr(_re.search(r'term7:\s*\[([^\]]*)\]', raw).group(1))
                term1 = _parse_num_arr(_re.search(r'term1:\s*\[([^\]]*)\]', raw).group(1))
                term14 = _parse_num_arr(_re.search(r'term14:\s*\[([^\]]*)\]', raw).group(1))
                
                # 构建操作列表并计算到期
                ops_list = []
                for i, d in enumerate(dates):
                    dt = _dt.strptime(d, '%Y-%m-%d')
                    t1v = term1[i] if i < len(term1) and term1[i] else 0
                    t7v = term7[i] if i < len(term7) and term7[i] else 0
                    t14v = term14[i] if i < len(term14) and term14[i] else 0
                    if t1v > 0: ops_list.append((dt, 1, t1v))
                    if t7v > 0: ops_list.append((dt, 7, t7v))
                    if t14v > 0: ops_list.append((dt, 14, t14v))
                
                # 计算到期日映射
                maturity_map = _dd(float)
                for dt, term, amount in ops_list:
                    maturity_map[dt + _td(days=term)] += amount
                
                # 找最近一个操作日
                op_dates = sorted(set(dt for dt, _, _ in ops_list))
                if op_dates:
                    latest_date = op_dates[-1]
                    total_ops = sum(a for d, t, a in ops_list if d == latest_date)
                    total_mats = maturity_map.get(latest_date, 0)
                    net_inj = total_ops - total_mats
                    
                    # 提取未到期余额
                    bal_match = _re.search(r'逆回购未到期余额.*?(\d[\d,]*)\s*亿元', html, _re.DOTALL)
                    outstanding = None
                    if bal_match:
                        outstanding = float(bal_match.group(1).replace(',', ''))
                    
                    # 提取MLF最新操作
                    mlf_match = _re.search(r'var mlfData\s*=\s*\{(.*?)\};\s*\n', html, _re.DOTALL)
                    mlf_latest = None
                    if mlf_match:
                        mlf_raw = mlf_match.group(1)
                        mlf_dates = _parse_str_arr(_re.search(r'dates:\s*\[([^\]]*)\]', mlf_raw).group(1))
                        mlf_amounts = _parse_num_arr(_re.search(r'amount:\s*\[([^\]]*)\]', mlf_raw).group(1))
                        for mi in range(len(mlf_dates)-1, -1, -1):
                            if mlf_amounts[mi] and mlf_amounts[mi] > 0:
                                mlf_latest = {'date': mlf_dates[mi], 'amount': mlf_amounts[mi]}
                                break
                    
                    ops_detail = {}
                    for d, t, a in ops_list:
                        if d == latest_date:
                            label = {1: '隔夜', 7: '7天', 14: '14天'}.get(t, f'{t}天')
                            ops_detail[label] = a
                    
                    print(f"  ✅ OMO(美元环流): {latest_date.strftime('%Y-%m-%d')} 净投放{net_inj:+.0f}亿 | 余额{outstanding:.0f}亿")
                    return {
                        'net_injection': net_inj,
                        'outstanding_balance': outstanding,
                        'operations': ops_detail,
                        'omo_date': latest_date.strftime('%Y-%m-%d'),
                        'mlf_latest': mlf_latest,
                        'total_ops': total_ops,
                        'total_maturities': total_mats,
                    }
    except Exception as e:
        print(f"  ⚠️ 美元环流OMO获取失败: {e}")
    
    print("  ⚠️ OMO数据缺失，显示⚠️")
    return {
        'net_injection': None,
        'outstanding_balance': None,
        'operations': {},
    }

    # ETF 6位代码列表
    etf_codes = [c.split('.')[0] for c in ETF_300_CODES]  # ['510300', '159919', '510310', '159922']
    
    try:
        import pkgutil, sys
        sys.modules.setdefault('pkgutil.ImpImporter', type('ImpImporter', (), {'find_module': lambda *a: None})())
        import akshare as ak
        df = ak.fund_etf_spot_em()
        if df is None or df.empty:
            print("  ⚠️ fund_etf_spot_em 返回空数据")
            return {'total_share': 0, 'change': None}
        
        # 逐只ETF取最新份额
        for code in etf_codes:
            row = df[df['代码'] == code]
            if not row.empty:
                share_val = _safe_float(row.iloc[0].get('最新份额'))
                if share_val:
                    total_share += share_val / 1e8  # 份 → 亿份
        
        print(f"  ✅ 300ETF总份额: {total_share:.1f}亿份")
        
        # 份额变化需历史数据 — 用data_hub兜底
        prev_share = 0.0
        for code in ETF_300_CODES:
            try:
                df_hist = get_etf_share(ts_code=code, lookback_days=5)
                if df_hist is not None and not df_hist.empty and len(df_hist) >= 2:
                    prev_row = df_hist.iloc[-2]
                    p_share = _safe_float(prev_row.get('fund_share'))
                    if p_share:
                        prev_share += p_share / 1e8
            except Exception:
                pass
        
        change = None
        if prev_share > 0:
            change = total_share - prev_share
        
        return {
            'total_share': total_share,
            'change': change,  # 亿份
        }
    except Exception as e:
        print(f"  ❌ ETF份额获取失败: {e}")
        return {'total_share': 0, 'change': None}

def fetch_margin(trade_date: str) -> Dict:
    """获取两融余额"""
    try:
        df = get_margin_trading(trade_date=trade_date, lookback_days=5)
        if df is None or df.empty:
            return {}
        
        latest = df.iloc[-1]
        rzye = _safe_float(latest.get('rzye'))  # tushare返回单位:元
        # 将元转换为亿元
        if rzye and rzye > 1e6:  # 大于100万说明单位是元
            rzye = rzye / 1e8
        
        # 前一日
        prev_rzye = None
        margin_change = None
        if len(df) >= 2:
            prev = df.iloc[-2]
            prev_rzye = _safe_float(prev.get('rzye'))
            if prev_rzye and prev_rzye > 1e6:
                prev_rzye = prev_rzye / 1e8
            if rzye and prev_rzye:
                margin_change = rzye - prev_rzye
        
        return {
            'rzye': rzye,
            'prev_rzye': prev_rzye,
            'change': margin_change,
        }
    except Exception as e:
        print(f"  ❌ 两融获取失败: {e}")
        return {}

        'iv_value': f"{iv_val:.1f}" if iv_val else "⚠️",
        'iv_zone': iv_zone,
        'iv_signal': iv_signal,
    }
    
    # 要素六：两年美债
    us_2y = global_data.get('us_2y')
    us_2y_change = ""  # 需要历史数据计算，暂时留空
    us_2y_data = {
        'rate': f"{us_2y:.2f}" if us_2y else "⚠️",
        'change': us_2y_change,
    }
    
    return {
        'increment_fund': increment_fund,
        'tech_sector': tech_sector,
        'central_bank': central_bank,
        'main_fund': main_fund,
        'iv_vix': iv_vix,
        'us_2y': us_2y_data,
    }


def analyze_size_style(size_data: Dict) -> Dict:
    """模块3：大小盘分析"""
    ratio = size_data.get('ratio')
    chg_5d = size_data.get('ratio_change_5d')
    chg_20d = size_data.get('ratio_change_20d')
    
    # 风格趋势判断
    if chg_5d is not None and chg_20d is not None:
        if chg_5d > 0 and chg_20d > 0:
            trend = "扩张"
            label = "小盘占优"
        elif chg_5d < 0 and chg_20d < 0:
            trend = "收窄"
            label = "大盘占优"
        elif chg_5d > 0:
            trend = "短期扩张"
            label = "小盘短期占优"
        else:
            trend = "短期收窄"
            label = "大盘短期占优"
    else:
        trend = "⚠️"
        label = "数据不足"
    
    return {
        'ratio': ratio,
        'ratio_change_5d': chg_5d,
        'ratio_change_20d': chg_20d,
        'style_trend': trend,
        'style_label': label,
    }


def analyze_bond(
    cd_data: Dict,
    bond_yields: Dict,
    tl_data: Dict,
    repo_data: Dict,
) -> Dict:
    """模块4：债市分析"""
    # 期限利差 = 10Y国债 - 1Y存单
    bond_10y = bond_yields.get('10Y')
    cd_rate = cd_data.get('rate')
    
    term_spread = None
    spread_status = "⚠️"
    if bond_10y and cd_rate:
        term_spread = (bond_10y - cd_rate) * 100  # bp
        if term_spread > 50:
            spread_status = "偏陡（内需较好）"
        elif term_spread > 25:
            spread_status = "正常偏陡"
        elif term_spread > 0:
            spread_status = "正常偏平"
        elif term_spread > -20:
            spread_status = "偏平（内需较差）"
        else:
            spread_status = "严重倒挂"
    
    # TL合约
    tl_ratio = tl_data.get('up_ratio_20d')
    if tl_ratio is not None:
        tl_zone = "配置区间（看多）" if tl_ratio >= 0.5 else "配置价值下降"
    else:
        tl_zone = "⚠️"
    
    # 资金面
    gc001 = repo_data.get('gc001')
    gc007 = repo_data.get('gc007')
    if gc001 is not None:
        if gc001 < 2.0:
            fund_status = "宽松"
        elif gc001 < 3.0:
            fund_status = "中性"
        else:
            fund_status = "紧张"
    else:
        fund_status = "⚠️"
    
    return {
        'term_spread': term_spread,
        'spread_status': spread_status,
        'tl_up_ratio': tl_ratio,
        'tl_config_zone': tl_zone,
        'gc001': gc001,
        'gc007': gc007,
        'fund_status': fund_status,
        'bond_yields': bond_yields,
    }


def analyze_market_status(
    index_data: Dict,
    margin_data: Dict,
    etf_data: Dict,
    size_data: Dict,
    iv_val: Optional[float],
) -> Dict:
    """模块5：市场状态判断（核心！）
    
    基于补跌原理判断：
    - 价值板块（沪深300）是否完成补跌 → vs 20日均线
    - 小市值板块（中证1000/2000）是否完成补跌 → vs 20日均线
    - 科技/成长（创业板/科创50）是否完成补跌 → vs 20日均线
    - 流动性剥离信号：两融大幅下降？主力资金大幅净卖出？
    """
    detail_lines = []
    
    # === 价值板块补跌判断 ===
    hs300 = index_data.get('沪深300', {})
    zz500 = index_data.get('中证500', {})
    
    value_above_ma20 = True  # 默认
    value_notes = []
    for name, info in [('沪深300', hs300), ('中证500', zz500)]:
        close = info.get('close')
        ma20 = info.get('ma20')
        if close and ma20:
            above = close > ma20
            value_notes.append(f"{name}{'站上' if above else '跌破'}20日均线")
            if not above:
                value_above_ma20 = False
        else:
            value_notes.append(f"{name}均线数据缺失")
    
    value_complete = value_above_ma20
    value_note = "；".join(value_notes)
    
    # === 小市值板块补跌判断 ===
    zz1000 = index_data.get('中证1000', {})
    small_close = zz1000.get('close')
    small_ma20 = zz1000.get('ma20')
    
    if small_close and small_ma20:
        small_above = small_close > small_ma20
        small_complete = small_above
        small_note = f"中证1000{'站上' if small_above else '跌破'}20日均线"
    else:
        small_complete = "⚠️"
        small_note = "数据缺失"
    
    # === 科技/成长板块补跌判断 ===
    cyb = index_data.get('创业板指', {})
    kcb = index_data.get('科创50', {})
    
    tech_above_ma20 = True
    tech_notes = []
    for name, info in [('创业板指', cyb), ('科创50', kcb)]:
        close = info.get('close')
        ma20 = info.get('ma20')
        if close and ma20:
            above = close > ma20
            tech_notes.append(f"{name}{'站上' if above else '跌破'}20日均线")
            if not above:
                tech_above_ma20 = False
        else:
            tech_notes.append(f"{name}均线数据缺失")
    
    tech_complete = tech_above_ma20
    tech_note = "；".join(tech_notes)
    
    # === 流动性剥离信号 ===
    # 两融大幅下降
    margin_chg = margin_data.get('change')
    leverage_stripping = False
    if margin_chg is not None and margin_chg < -400:
        leverage_stripping = True
        leverage_note = f"单日两融下降{abs(margin_chg):.0f}亿（>400亿=剧烈去杠杆）"
    else:
        leverage_note = f"两融变动{margin_chg:+.0f}亿" if margin_chg else "数据不足"
    
    # 主力资金大幅净卖出
    etf_change = etf_data.get('change')
    fund_outflow = False
    if etf_change is not None and etf_change < -10:
        fund_outflow = True
        fund_note = f"ETF份额减少{abs(etf_change):.1f}亿份"
    else:
        fund_note = f"ETF份额变动{etf_change:+.1f}亿份" if etf_change else "数据不足"
    
    # === 综合判断 ===
    # 底部确认：价值+小市值都完成补跌，无流动性剥离
    # 补跌进行中：任一板块仍在补跌且流动性剥离
    # 边际观察：中间状态
    
    detail_lines.append(f"价值板块: {value_note}")
    detail_lines.append(f"小市值板块: {small_note}")
    detail_lines.append(f"科技/成长: {tech_note}")
    detail_lines.append(f"两融变动: {leverage_note}")
    detail_lines.append(f"ETF变动: {fund_note}")
    
    if value_complete and small_complete and tech_complete:
        overall = "底部确认"
        overall_emoji = "🔵"
    elif not value_complete and leverage_stripping:
        overall = "补跌进行中"
        overall_emoji = "🔴"
    elif not value_complete or not small_complete:
        # 有板块未完成但未剧烈去杠杆
        if small_complete:
            overall = "边际观察"
            overall_emoji = "🟡"
        else:
            overall = "补跌进行中"
            overall_emoji = "🔴"
    else:
        # 价值和小盘完成，科技未完成
        overall = "边际观察"
        overall_emoji = "🟡"
    
    # IV辅助判断
    if iv_val and iv_val > 25:
        detail_lines.append(f"IV={iv_val:.1f}%>25%，恐慌加剧，确认补跌信号")
    elif iv_val and iv_val < 15:
        detail_lines.append(f"IV={iv_val:.1f}%<15%，波动率极低，市场处于平静期")
    
    return {
        'value_complete': value_complete,
        'value_note': value_note,
        'small_cap_complete': small_complete,
        'small_cap_note': small_note,
        'tech_complete': tech_complete,
        'tech_note': tech_note,
        'leverage_stripping': leverage_stripping,
        'leverage_note': leverage_note,
        'fund_outflow': fund_outflow,
        'fund_note': fund_note,
        'overall': overall,
        'overall_emoji': overall_emoji,
        'detail_lines': detail_lines,
    }


def analyze_advice(status_data: Dict, size_data: Dict, iv_val: Optional[float]) -> Dict:
    """模块6：操作建议"""
    overall = status_data.get('overall', '')
    leverage = status_data.get('leverage_stripping', False)
    small_complete = status_data.get('small_cap_complete', False)
    
    suggestions = []
    risk_warning = ""
    
    if '底部' in overall:
        status_label = "底部确认"
        status_emoji = "🔵"
        suggestions.append("自有资金账户：可适度加仓，优先价值板块")
        suggestions.append("杠杆账户：控制仓位，确认右侧信号再加杠杆")
        suggestions.append("科技股：等待K线站上20日均线后再参与")
    elif '边际' in overall:
        status_label = "边际观察"
        status_emoji = "🟡"
        suggestions.append("自有资金账户：耐心等待右侧信号（小市值指数站上20日均线）")
        suggestions.append("杠杆账户：控制仓位，算好强平线，准备预备队补保证金")
        suggestions.append("科技股：尊重K线，尤其是20日均线，等待右侧")
        if not small_complete:
            risk_warning = "小市值指数未确认底部，补跌可能继续"
    elif '补跌' in overall:
        status_label = "补跌进行中"
        status_emoji = "🔴"
        suggestions.append("自有资金账户：保持审慎，等待补跌完成")
        suggestions.append("杠杆账户：减仓或对冲，优先保护本金")
        suggestions.append("科技股：不抄底，尊重20日均线压制")
        if leverage:
            risk_warning = "剧烈去杠杆中，优先保护本金，千万算好强平线"
        else:
            risk_warning = "补跌尚未完成，等待小市值指数底部信号"
    else:
        status_label = "⚠️ 数据不足"
        status_emoji = "⚪"
        suggestions.append("数据不足以做出明确判断，请手动补充分析")
    
    # 大小盘风格建议
    ratio_chg = size_data.get('ratio_change_5d')
    if ratio_chg is not None:
        if ratio_chg > 2:
            suggestions.append("大小盘：小盘短期偏强，关注小市值增量资金持续性")
        elif ratio_chg < -2:
            suggestions.append("大小盘：大盘短期偏强，价值风格修复中")
    
    # IV建议
    if iv_val and iv_val > 23:
        suggestions.append(f"波动率：IV={iv_val:.1f}%>23%，控制仓位，防范波动率放大")
    
    return {
        'status_label': status_label,
        'status_emoji': status_emoji,
        'suggestions': suggestions,
        'risk_warning': risk_warning,
    }


# ===== 主流程 =====

def run_analysis(trade_date: str, skip_push: bool = False) -> Dict:
    """运行完整分析流程"""
    print(f"\n{'='*60}")
    print(f"🐕 土狗分析日报 - {trade_date}")
    print(f"{'='*60}\n")
    
    # 1. 幂等检查
    _init_state_db()
    if _is_cached(trade_date):
        print(f"✅ {trade_date} 已推送，跳过（幂等）")
        return {"status": "cached", "trade_date": trade_date}
    
    # 2. 数据获取
    print("📊 [1/6] 获取指数行情数据...")
    index_data = fetch_index_data(trade_date, lookback=60)
    
    print("\n📊 [2/6] 获取利率与债市数据...")
    cd_data = fetch_cd_rate(trade_date)
    bond_yields = fetch_bond_yield()
    
    print("\n📊 [3/6] 获取央行与海外数据...")
    omo_data = fetch_omo(trade_date)
    global_data = fetch_global_macro(trade_date)
    repo_data = fetch_repo(trade_date)
    tl_data = fetch_treasury_futures()
    
    print("\n📊 [4/6] 获取市场微观数据...")
    size_data_raw = fetch_size_style()
    etf_data = fetch_etf_share()
    margin_data = fetch_margin(trade_date)
    
    print("\n📊 [5/6] 获取期权IV...")
    iv_val = fetch_option_iv(trade_date)
    
    print("\n📊 [6/7] 获取板块资金流...")
    sector_data = get_sector_flow()
    
    print("\n🤖 [7/7] AI综合研判...")
    # 先做分析计算（AI需要这些结果）
    indicators = analyze_indicators(cd_data, bond_yields, global_data, omo_data, iv_val, margin_data, etf_data)
    factors = analyze_factors(index_data, cd_data, omo_data, etf_data, iv_val, global_data)
    size_analysis = analyze_size_style(size_data_raw)
    bond_analysis = analyze_bond(cd_data, bond_yields, tl_data, repo_data)
    status_result = analyze_market_status(index_data, margin_data, etf_data, size_data_raw, iv_val)
    advice_result = analyze_advice(status_result, size_data_raw, iv_val)
    
    # AI研判
    ai_text = fetch_ai_analysis(
        cd_data, bond_yields, global_data, omo_data, iv_val,
        margin_data, etf_data, status_result, size_analysis,
    )
    
    print("\n🧮 开始分析计算...\n")
    
    # 3. 分析计算已在上方完成（AI研判需要先计算结果）
    
    # 4. 指数行情摘要
    index_summary = {}
    for name, info in index_data.items():
        index_summary[name] = {
            'change_pct': info.get('change_pct'),
            'volume_change': info.get('vol_change', ''),
            'ma20_position': info.get('ma20_position', ''),
        }
    
    # 5. 构建卡片
    date_str = f"{trade_date[:4]}-{trade_date[4:6]}-{trade_date[6:8]}"
    
    card = build_tugou_analysis_card(
        date_str=date_str,
        indicators=indicators,
        factors=factors,
        size_data=size_analysis,
        bond_data=bond_analysis,
        status_data=status_result,
        advice_data=advice_result,
        index_summary=index_summary,
        ai_analysis=ai_text,
    )
    
    card_json = json.dumps(card, ensure_ascii=False)
    
    # 6. 推送
    if not skip_push:
        print(f"\n📤 推送飞书卡片到群 {CHAT_ID[:10]}...")
        ok = push_card_to_feishu(card_dict=card, chat_id=CHAT_ID)
        if ok:
            print("✅ 推送成功")
        else:
            print("❌ 推送失败，仍保存缓存")
    else:
        print("\n⏭️ 跳过推送（--skip-push）")
    
    # 7. 保存缓存
    _save_cache(trade_date, card_json)
    
    # 8. 输出摘要
    print(f"\n{'='*60}")
    print(f"📋 分析摘要:")
    print(f"  市场状态: {status_result.get('overall_emoji', '')} {status_result.get('overall', '')}")
    print(f"  大小盘: {size_analysis.get('style_trend', '')} → {size_analysis.get('style_label', '')}")
    if iv_val:
        print(f"  IV: {iv_val:.1f}% ({_iv_zone(iv_val)[0]})")
    print(f"{'='*60}\n")
    
    return {
        "status": "success",
        "trade_date": trade_date,
        "overall": status_result.get('overall', ''),
        "card_size": len(card_json),
    }


def main():
    parser = argparse.ArgumentParser(description="土狗分析日报")
    parser.add_argument("trade_date", nargs="?", default=None,
                        help="交易日期 YYYYMMDD，默认最近交易日")
    parser.add_argument("--skip-push", action="store_true",
                        help="跳过飞书推送（仅本地运行）")
    parser.add_argument("--force", action="store_true",
                        help="强制重新运行（忽略幂等缓存）")
    args = parser.parse_args()
    
    # 确定交易日期
    if args.trade_date:
        # 校验格式，防止传入非日期参数（如 no_reply）导致崩溃
        try:
            datetime.strptime(args.trade_date, "%Y%m%d")
            trade_date = args.trade_date
        except ValueError:
            print(f"  ⚠️ trade_date 参数 '{args.trade_date}' 格式非法，使用默认最近交易日")
            trade_date = None
    if not args.trade_date or trade_date is None:
        # 获取最近交易日
        try:
            today = datetime.now().strftime("%Y%m%d")
            # 简单回退找交易日
            td = datetime.now()
            for i in range(5):
                check_date = (td - timedelta(days=i)).strftime("%Y%m%d")
                # 周末跳过
                check_td = datetime.strptime(check_date, "%Y%m%d")
                if check_td.weekday() < 5:
                    trade_date = check_date
                    break
            else:
                trade_date = today
        except Exception:
            trade_date = datetime.now().strftime("%Y%m%d")
    
    # 强制模式：删除缓存
    if args.force and STATE_DB.exists():
        conn = sqlite3.connect(str(STATE_DB))
        conn.execute("DELETE FROM tugou_daily_cache WHERE trade_date=?", (trade_date,))
        conn.commit()
        conn.close()
        print(f"🔄 已清除 {trade_date} 的缓存（--force）")
    
    result = run_analysis(trade_date, skip_push=args.skip_push)
    
    # CodeAct提交结果
    try:
        sys.path.insert(0, './codeact/scripts')
        from codeact_sdk import CodeActSDK
        sdk = CodeActSDK()
        
        overall = result.get('overall', '')
        if '底部' in overall:
            msg_prefix = "🟢"
        elif '补跌' in overall:
            msg_prefix = "🔴"
        else:
            msg_prefix = "🟡"
        
        msg = (
            f"{msg_prefix} 土狗分析日报 - {trade_date}\n"
            f"市场状态: {result.get('overall', 'N/A')}\n"
            f"{'已推送飞书群' if not args.skip_push else '未推送（--skip-push）'}"
        )
        
        sdk.submit_result(
            status="success",
            result_mode="display_only",
            message=msg,
            data=result,
        )
    except Exception as e:
        # 不在CodeAct环境中运行时，直接打印
        print(f"\n✅ 分析完成: {result}")
        print(f"(CodeAct SDK不可用: {e})")


if __name__ == '__main__':
    main()
