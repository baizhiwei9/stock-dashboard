
(function(){
  "use strict";

  /* ---------- 工具 ---------- */
  function fmt(v,dec){
    if(v===null||v===undefined||isNaN(v)) return "—";
    dec = (dec===undefined?2:dec);
    return Number(v).toFixed(dec);
  }
  function tickText(v,r){
    if(Math.abs(v)>=1000) return String(Math.round(v));
    var dec = r>=20?0 : r>=3?1 : r>=0.15?2 : 3;
    return v.toFixed(dec);
  }
  function signCls(v){
    if(v===null||v===undefined||isNaN(v)) return "flat";
    if(v>0) return "up"; if(v<0) return "down"; return "flat";
  }
  function pctCls(p){
    if(p===null||p===undefined) return "flat";
    if(p>=90) return "c-red"; if(p>=70) return "c-amber"; if(p<=10) return "c-green"; return "flat";
  }

  /* ---------- SVG 折线图 ---------- */
  var NS="http://www.w3.org/2000/svg";
  function el(tag,attrs){
    var e=document.createElementNS(NS,tag);
    for(var k in attrs){ e.setAttribute(k,attrs[k]); }
    return e;
  }
  function drawChart(host,cfg){
    host.innerHTML="";
    var labels=cfg.labels||[], series=(cfg.series||[]).filter(function(s){return s.values&&s.values.some(function(v){return v!==null&&v!==undefined;});});
    if(!labels.length||!series.length){ host.innerHTML='<p class="note">无数据</p>'; return; }
    var W=800,H=cfg.height||250;
    var P={t:14,r:12,b:24,l:cfg.padL||46};
    var iw=W-P.l-P.r, ih=H-P.t-P.b;

    var all=[];
    series.forEach(function(s){ s.values.forEach(function(v){ if(v!==null&&v!==undefined&&!isNaN(v)) all.push(v); }); });
    if(!all.length){ host.innerHTML='<p class="note">无数据</p>'; return; }
    var lo=Math.min.apply(null,all), hi=Math.max.apply(null,all);
    if(cfg.min!==undefined) lo=Math.min(lo,cfg.min);
    if(cfg.max!==undefined) hi=Math.max(hi,cfg.max);
    if(hi===lo){ hi=lo+1; lo=lo-1; }
    var padV=(hi-lo)*0.12; lo-=padV; hi+=padV;
    if(cfg.zeroBase){ lo=Math.min(lo,0); }
    if(cfg.yMin!==undefined) lo=cfg.yMin;
    if(cfg.yMax!==undefined) hi=cfg.yMax;

    function X(i){ return P.l + (labels.length===1?iw/2:iw*i/(labels.length-1)); }
    function Y(v){ return P.t + ih*(1-(v-lo)/(hi-lo)); }

    var svg=el("svg",{viewBox:"0 0 "+W+" "+H,role:"img","aria-label":cfg.aria||"走势图"});
    var title=el("title",{}); title.textContent=cfg.aria||"走势图"; svg.appendChild(title);

    /* 网格 + Y 轴刻度 */
    var TICKS=4;
    for(var t=0;t<=TICKS;t++){
      var val=lo+(hi-lo)*t/TICKS, y=Y(val);
      svg.appendChild(el("line",{x1:P.l,x2:W-P.r,y1:y,y2:y,stroke:"#232c3b","stroke-width":1}));
      var tx=el("text",{x:P.l-7,y:y+3.5,"text-anchor":"end",fill:"#6b7889","font-size":10.5,"font-family":"ui-monospace,monospace"});
      tx.textContent=cfg.yfmt?cfg.yfmt(val):tickText(val,hi-lo);
      svg.appendChild(tx);
    }
    /* 零轴 */
    if(lo<0&&hi>0){
      svg.appendChild(el("line",{x1:P.l,x2:W-P.r,y1:Y(0),y2:Y(0),stroke:"#3a4658","stroke-width":1,"stroke-dasharray":"3 3"}));
    }
    /* X 轴标签 */
    var step=Math.max(1,Math.ceil(labels.length/6));
    for(var i=0;i<labels.length;i+=step){
      var xt=el("text",{x:X(i),y:H-7,"text-anchor":i===0?"start":(i>=labels.length-step?"end":"middle"),fill:"#6b7889","font-size":10.5,"font-family":"ui-monospace,monospace"});
      xt.textContent=String(labels[i]).slice(5);
      svg.appendChild(xt);
    }
    /* 折线 */
    series.forEach(function(s){
      var d="",open=false;
      s.values.forEach(function(v,i){
        if(v===null||v===undefined||isNaN(v)){ open=false; return; }
        d+=(open?"L":"M")+X(i).toFixed(1)+" "+Y(v).toFixed(1)+" ";
        open=true;
      });
      if(d){
        var p=el("path",{d:d,fill:"none",stroke:s.color,"stroke-width":1.8,"stroke-linejoin":"round","stroke-linecap":"round",opacity:s.dim?0.35:1});
        svg.appendChild(p);
      }
    });

    /* hover */
    var hov=el("g",{opacity:"0"});
    var vline=el("line",{y1:P.t,y2:P.t+ih,stroke:"#5b6b82","stroke-width":1,"stroke-dasharray":"3 3"});
    hov.appendChild(vline);
    var dots=series.map(function(s){
      var c=el("circle",{r:3.2,fill:s.color,stroke:"#0b0e14","stroke-width":1.2});
      hov.appendChild(c); return c;
    });
    svg.appendChild(hov);
    var hit=el("rect",{x:P.l,y:P.t,width:iw,height:ih,fill:"transparent",style:"cursor:crosshair"});
    svg.appendChild(hit);

    var tip=document.createElement("div"); tip.className="tip";

    function move(ev){
      var box=svg.getBoundingClientRect();
      var px=(ev.clientX-box.left)/box.width*W;
      var i=Math.round((px-P.l)/(iw||1)*(labels.length-1));
      i=Math.max(0,Math.min(labels.length-1,i));
      hov.setAttribute("opacity","1");
      vline.setAttribute("x1",X(i)); vline.setAttribute("x2",X(i));
      var rows="";
      series.forEach(function(s,j){
        var v=s.values[i];
        var has=(v!==null&&v!==undefined&&!isNaN(v));
        dots[j].setAttribute("opacity",has?"1":"0");
        if(has){ dots[j].setAttribute("cx",X(i)); dots[j].setAttribute("cy",Y(v)); }
        rows+='<div class="r"><span style="color:'+s.color+'">'+s.name+'</span><span>'+(has?(cfg.vfmt?cfg.vfmt(v):fmt(v)):"—")+(cfg.unit||"")+'</span></div>';
      });
      tip.innerHTML='<div class="d">'+labels[i]+'</div>'+rows;
      tip.classList.add("on");
      var bw=box.width, tw=tip.offsetWidth;
      var left=(X(i)/W)*bw+12;
      if(left+tw>bw-8){ left=(X(i)/W)*bw-tw-12; }
      tip.style.left=Math.max(0,left)+"px";
      tip.style.top=Math.max(0,(Y(0)/H)*box.height-10)+"px";
    }
    hit.addEventListener("mousemove",move);
    hit.addEventListener("mouseleave",function(){ hov.setAttribute("opacity","0"); tip.classList.remove("on"); });
    hit.addEventListener("touchstart",function(e){ if(e.touches[0]) move(e.touches[0]); },{passive:true});
    hit.addEventListener("touchmove",function(e){ if(e.touches[0]) move(e.touches[0]); },{passive:true});

    host.appendChild(svg); host.appendChild(tip);
  }

  function legend(items){
    return items.map(function(s){
      return '<span><b style="background:'+s.color+'"></b>'+s.name+'</span>';
    }).join("");
  }
